window.sessionRedeem = async () => { const item = sessionList[sessionCurrentIndex]; try { // Nếu chưa có location (trường hợp dùng đơn lẻ) → lấy lúc này if (!sessionLocation) { showLoading("Đang lấy vị trí..."); sessionLocation = await getCurrentLocation(); hideLoading(); } const location = sessionLocation || "Không xác định"; await updateDoc(doc(db, "vouchers", item.id), { is_used: true, used_at: new Date().toISOString(), used_location: location }); sessionRedeemedAmount += (item.amount || 0); updateRedeemedAmountDisplay(); // Animation thành công await playRedeemSuccessAnimation(); sessionList.splice(sessionCurrentIndex, 1); document.getElementById('sessionTotal').innerText = sessionList.length; document.getElementById('swipeHint').classList.toggle('hidden', sessionList.length <= 1); if (sessionList.length === 0) { // Voucher cuối cùng → toast tổng tiền showToast(`Đã dùng ${formatCurrency(sessionRedeemedAmount)} trong phiên này`); endSession(); } else { if (sessionCurrentIndex >= sessionList.length) sessionCurrentIndex = sessionList.length - 1; renderSessionItem(); } } catch (err) { alert("Lỗi khi đánh dấu sử dụng: " + err.message); } };
